cmake_minimum_required(VERSION 3.15)
project(FileOrganizer CXX)

set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)

if(NOT CMAKE_BUILD_TYPE AND NOT CMAKE_CONFIGURATION_TYPES)
    set(CMAKE_BUILD_TYPE Release)
endif()

find_package(Threads REQUIRED)

add_executable(organizer main.cpp organizer_engine.cpp)
target_include_directories(organizer PRIVATE ${CMAKE_SOURCE_DIR})
target_link_libraries(organizer PRIVATE Threads::Threads)

option(ORGANIZER_BUILD_TESTS "Build the unit test suite" ON)
if(ORGANIZER_BUILD_TESTS)
    enable_testing()
    add_executable(organizer_tests tests/test_main.cpp tests/test_core.cpp organizer_engine.cpp)
    target_include_directories(organizer_tests PRIVATE ${CMAKE_SOURCE_DIR})
    target_link_libraries(organizer_tests PRIVATE Threads::Threads)
    add_test(NAME organizer_tests COMMAND organizer_tests)
endif()

option(ORGANIZER_BUILD_GUI "Build the SDL3 GUI (organizer_gui)" ON)
if(ORGANIZER_BUILD_GUI)
    find_package(SDL3 QUIET)
    find_package(SDL3_ttf QUIET)
    if(SDL3_FOUND AND SDL3_ttf_FOUND)
        add_executable(organizer_gui gui/organizer_gui.cpp gui/widgets.cpp organizer_engine.cpp)
        target_include_directories(organizer_gui PRIVATE ${CMAKE_SOURCE_DIR})
        target_link_libraries(organizer_gui PRIVATE SDL3::SDL3 SDL3_ttf::SDL3_ttf Threads::Threads)
        if(WIN32)
            set_target_properties(organizer_gui PROPERTIES WIN32_EXECUTABLE TRUE)
        endif()
        add_custom_command(TARGET organizer_gui POST_BUILD
            COMMAND ${CMAKE_COMMAND} -E copy_directory
                    ${CMAKE_SOURCE_DIR}/assets $<TARGET_FILE_DIR:organizer_gui>/assets)
    else()
        message(STATUS "SDL3/SDL3_ttf not found - skipping organizer_gui (set ORGANIZER_BUILD_GUI=OFF to silence)")
        set(ORGANIZER_BUILD_GUI OFF)
    endif()
endif()
